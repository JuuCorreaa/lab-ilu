// Função para abrir o visualizador de apresentações/marcas diretamente no site function openBrandViewer(filePath, title) { const modal = document.getElementById('brand-viewer-modal'); const iframe = document.getElementById('brand-viewer-iframe'); const titleEl = document.getElementById('brand-viewer-title'); if (modal && iframe) { iframe.src = filePath; if (titleEl) titleEl.textContent = title; modal.classList.remove('hidden'); modal.classList.add('flex'); document.body.style.overflow = 'hidden'; // Impede o scroll de fundo } } // Função para fechar o visualizador function closeBrandViewer() { const modal = document.getElementById('brand-viewer-modal'); const iframe = document.getElementById('brand-viewer-iframe'); if (modal && iframe) { iframe.src = ''; modal.classList.add('hidden'); modal.classList.remove('flex'); document.body.style.overflow = ''; // Restaura o scroll } }
